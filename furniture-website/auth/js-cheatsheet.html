Everything starts with functions()

Explore declarations, parameters, scope & closures, this, higher-order utilities, async patterns, errors, generators, and more — each with a runnable demo.

1) Function Forms

Declaration · Expression · Arrow · IIFE
// Declaration
function add(a, b) { return a + b; }

// Expression
const multiply = function(a, b) { return a * b; }

// Arrow (concise)
const square = x => x * x;

// IIFE (executes immediately)
const START = (() => Date.now())();
Tip: use declarations for hoisting, arrows for concise callbacks.

2) Parameters

Default · Rest · Destructuring
// Default params
function greet(name = "friend") {
  return `Hi, ${name}!`;
}

// Rest params
function sumAll(...nums) {
  return nums.reduce((t, n) => t + n, 0);
}

// Destructuring params
function drawPoint({ x = 0, y = 0 }) {
  return `(${x}, ${y})`;
}

3) Scope & Closures

A closure preserves access to outer variables.
function makeCounter(start = 0) {
  let n = start;
  return function() {
    return ++n;
  };
}
const c1 = makeCounter(5);
// c1() -> 6, then 7, ...

4) this · bind / call / apply

const timer = {
  t: 0,
  tick() { this.t++; },
};
const inc = timer.tick;
// inc() alone loses this; bind fixes it:
const incBound = timer.tick.bind(timer);

5) Higher-Order Functions

Functions that take or return functions.
const nums = [1, 2, 3, 4, 5];
const doubled = nums.map(n => n * 2);
const evens  = nums.filter(n => n % 2 === 0);
const total  = nums.reduce((t, n) => t + n, 0);

6) Async: Callbacks → Promises → async/await

function wait(ms) {
  return new Promise(r => setTimeout(() => r(`Waited ${ms}ms`), ms));
}

async function fetchMessage() {
  const a = await wait(200);
  const b = await wait(100);
  return `${a} & ${b}`;
}

7) Error Handling

function mustBeNumber(x) {
  if (typeof x !== "number") throw new TypeError("Expected number");
  return x * 2;
}

function safe(fn) {
  try { return { ok: true, data: fn() }; }
  catch (e) { return { ok: false, error: e.message }; }
}

8) Generators & Iterators

function* range(a, b) {
  for (let i = a; i <= b; i++) yield i;
}
const it = range(3, 6); // lazily produces 3..6

9) Recursion & Memoization

const memo = (fn) => {
  const cache = new Map();
  return (...args) => {
    const k = JSON.stringify(args);
    if (cache.has(k)) return cache.get(k);
    const v = fn(...args);
    cache.set(k, v);
    return v;
  };
};
const fib = memo(function f(n) {
  return n < 2 ? n : f(n - 1) + f(n - 2);
});

10) Currying · Partial · Composition

const add = a => b => a + b;              // curried
const partial = (fn, ...a) => (...b) => fn(...a, ...b);
const compose = (...fns) => x => fns.reduceRight((v, f) => f(v), x);

const double = x => x * 2;
const inc    = x => x + 1;
const incThenDouble = compose(double, inc); // double(inc(x))
